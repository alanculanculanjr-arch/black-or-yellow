Round engine: settling a round

app/src/rounds.js, from line 134. When the banker confirms the colour, every locked bet is paid 2x or marked lost, inside one database transaction, so a round can never be paid twice.

  function settleInner(roundId, color, detectedColor) {
    if (q.settle.run(color, color !== detectedColor ? 1 : 0, roundId).changes !== 1) return false;
    for (const bet of q.lockedBets.all(roundId)) {
      if (bet.color === color) {
        wallet.apply({
          playerId: bet.player_id, type: 'WIN_PAYOUT', amount: bet.amount * PAYOUT_MULTIPLIER, key: `payout:${bet.id}`, roundId,
        });
        q.setBet.run('WON', bet.id);
      } else {
        q.setBet.run('LOST', bet.id);
      }
    }
    log(`round ${roundId} settled: ${color}` + (color !== detectedColor ? ` (banker corrected from ${detectedColor})` : ''));
    return true;
  }

  // ---- slow steps (run after the transaction has committed) ----------------------------

Round engine: placing a bet

app/src/rounds.js, from line 402. Checks the round is open, the player is active, the amount is 5/10/15/20 and affordable, then deducts the points.

  const placeBet = db.transaction((playerId, { roundId, color, amount } = {}) => {
    const { g, round } = current();
    if (g.state !== 'BETTING_OPEN') return fail('BETTING_CLOSED', 'Betting is closed.');
    if (roundId !== undefined && Number(roundId) !== round.id) {
      return fail('ROUND_CHANGED', 'That round has ended. Check the current round and try again.');
    }
    const p = q.player.get(playerId);
    if (!p || p.status !== 'ACTIVE') return fail('NOT_ACTIVE', 'Only active players can bet.');
    if (p.leave_requested_at) return fail('LEAVING', 'You are leaving the game and cannot place more bets.');
    if (!COLORS.includes(color)) return fail('BAD_COLOR', 'Choose BLACK or YELLOW.');
    if (!BET_AMOUNTS.includes(amount)) return fail('BAD_AMOUNT', 'Bets are 5, 10, 15 or 20 points.');
    const existing = q.betFor.get(round.id, playerId);
    if (existing) return fail('ALREADY_BET', 'You already have a bet in this round.');
    if (amount > p.balance - p.reserved) return fail('INSUFFICIENT', 'Not enough points for that bet.');

    const betId = q.insertBet.run(round.id, playerId, color, amount).lastInsertRowid;
    wallet.apply({ playerId, type: 'BET_DEBIT', amount: -amount, key: `bet:${betId}`, roundId: round.id });
    log(`player ${playerId} bet ${amount} on ${color} in round ${round.id}`);
    return { ok: true, bet: { roundId: round.id, color, amount, status: 'LOCKED' } };
  }).immediate;

Wallet: the only code that changes balances

app/src/wallet.js, from line 32. Every change writes one ledger row with a unique key; the same event (one payout, one approval) can be recorded only once.

  function apply({ playerId, type, amount, key, requestId = null, roundId = null }) {
    if (!db.inTransaction) throw new Error('wallet.apply must run inside a transaction');
    if (!Number.isInteger(amount) || amount === 0) throw new Error(`bad amount ${amount}`);
    if (q.hasKey.get(key)) return { applied: false, duplicate: true };
    const p = q.player.get(playerId);
    if (!p) throw new Error(`no player ${playerId}`);
    const after = p.balance + amount;
    if (after < p.reserved) throw new Error(`player ${playerId} would go below reserved points`);
    q.insert.run(playerId, type, amount, after, key, requestId, roundId);
    q.setBalance.run(after, playerId);
    return { applied: true, balance: after };
  }

  // Holding points for a pending redemption doesn't move them, so it isn't a ledger entry;
  // the reserved total is checked against pending redemptions by audit() instead.

Point Store: banker decisions

app/src/points.js, from line 94. Approve, reject or cancel a top-up or redemption; a second click finds the request already handled and changes nothing.

  function decideInner(requestId, action, by) {
    const req = q.byId.get(requestId);
    if (!req) return fail('NOT_FOUND');
    if (req.status !== 'PENDING') return fail('NOT_PENDING', { status: req.status });

    const status = { approve: 'APPROVED', reject: 'REJECTED', cancel: 'CANCELLED' }[action];
    if (!status) throw new Error(`bad action ${action}`);
    if (q.resolve.run(status, by, requestId).changes !== 1) return fail('NOT_PENDING');

    if (req.type === 'REDEMPTION') wallet.release(req.player_id, req.amount);
    if (status === 'APPROVED') {
      wallet.apply({
        playerId: req.player_id,
        type: req.type === 'PURCHASE' ? 'POINT_PURCHASE' : 'POINT_REDEMPTION',
        amount: req.type === 'PURCHASE' ? req.amount : -req.amount,
        key: `request:${req.id}`,
        requestId: req.id,
      });
    }
    const done = q.byId.get(requestId);
    return { ok: true, request: done, playerId: done.player_id, notice: noticeFor(done) };
  }

Hardware sequence: drop and tilt

app/src/real.js, from line 21. DROP runs the relay for the configured seconds; recovery tilts the servo, waits, and levels the board.

    async dropBall() {
      log(`[hardware] relay ON for ${cfg.dropRelaySeconds}s (ball release)`);
      await w.call('relay_pulse', { seconds: cfg.dropRelaySeconds }, (cfg.dropRelaySeconds + 5) * 1000);
    },
    async recoverBall() {
      const my = ++epoch;
      log(`[hardware] tilt to ${cfg.tiltAngle}° for ${cfg.tiltHoldSeconds}s, then back to ${cfg.levelAngle}°`);
      await w.call('servo', { angle: cfg.tiltAngle });
      await sleep(cfg.tiltHoldSeconds * 1000);
      if (my !== epoch) return;
      await w.call('servo', { angle: cfg.levelAngle });
      await sleep(700);                       // let it arrive before letting go
      if (my !== epoch) return;
      await w.call('servo_release');
    },
    stopAll() {
      epoch++;
      w.call('all_off').catch((e) => log('[hardware] all_off failed:', e.message));

Hardware helper: relay and servo

hardware/hw_helper.py, from line 39. Drives GPIO 17 (relay, NO/NC aware) and GPIO 18 (servo pulse 1.0-2.0 ms).

def relay(on):
    """on=True: the DEVICE runs (whichever relay contacts it is wired to)."""
    with _gpio:
        lgpio.gpio_write(h, RELAY_PIN, RUN_LEVEL if on else IDLE_LEVEL)


def servo(angle):
    global servo_claimed
    angle = max(0.0, min(180.0, float(angle)))
    with _gpio:
        if not servo_claimed:
            lgpio.gpio_claim_output(h, SERVO_PIN, 0)
            servo_claimed = True
        lgpio.tx_servo(h, SERVO_PIN, int(MIN_US + (MAX_US - MIN_US) * angle / 180))
    return angle

Vision: colour share of the ball

vision/detector.py, from line 96. Adds up how much of the ball lies on black and on yellow cells; only the part over the board counts.

def colour_share(P):
    """(winning colour, its share, other share, cell with most of the ball) for touching point P (mm)."""
    s = 4
    n = int(A * s)
    disk = np.zeros((n, n), np.uint8)
    cv2.circle(disk, (int(P[0] * s), int(P[1] * s)), int(R_BALL * s), 1, -1)
    total = np.pi * (R_BALL * s) ** 2
    share = {'BLACK': 0.0, 'YELLOW': 0.0}
    best = (-1.0, 0)
    for cell, x0, y0 in CELLS:
        sh = disk[int(y0 * s):int((y0 + C) * s), int(x0 * s):int((x0 + C) * s)].sum() / total
        share['BLACK' if cell in BLACK else 'YELLOW'] += sh
        best = max(best, (sh, cell))
    on_board = share['BLACK'] + share['YELLOW']
    if on_board > 0:                               # edge rule: only the part over the board counts
        share = {c: v / on_board for c, v in share.items()}
    win = max(share, key=share.get)
    lose = 'YELLOW' if win == 'BLACK' else 'BLACK'
    return win, share[win], share[lose], best[1]

Vision worker: the 2-second still rule

vision/worker.py, from line 59. Watches the camera until the ball has stayed put for 2 seconds or the time limit passes.

def detect(timeout, still_s):
    try:
        grab(allow_search=True)               # camera moved since last time? find it before the clock starts
    except Exception:
        pass
    start = last_ok = time.time()
    tracker = detector.StillTracker(seconds=still_s)
    last_vis, last_info = None, {}
    while True:
        now = time.time()
        if now - start > timeout:
            text = f'TIMEOUT after {timeout:g}s: {last_info.get("result", "no picture")}'
            return {'status': 'TIMEOUT', 'last': last_info.get('result'),
                    'snapshot': snapshot(last_vis, text) if last_vis is not None else None}
        try:
            img = grab()
            last_ok = now
        except Exception as e:
            if time.time() - last_ok > CAMERA_LOST_S:
                return {'status': 'CAMERA_LOST', 'error': str(e)[:200]}
            time.sleep(0.2)
            continue
        vis, info = detector.find_ball(img)
        last_vis, last_info = vis, info
        if 'P' not in info:                     # no ball / board not found: stillness starts over
            tracker.reset()
            continue
        if tracker.add(info['P']):
            base = {'cell': info['cell'], 'shares': info['shares'], 'share_win': info['share_win'],
                    'share_lose': info['share_lose'], 'markers': info['markers'], 'edge': bool(info.get('edge'))}
